Как подключить модальное окно на любую страницу

Модуль vip-concierge-modal.js написан на чистом нативном JavaScript (Zero external framework dependencies) и адаптирован под институциональный стиль. Достаточно подключить один скрипт: 

<!-- 1. Подключение скрипта движка -->
<script src="vip-concierge-modal.js"></script>

<!-- 2. Вызов модального окна по клику на любую кнопку -->
<button onclick="VIPConcierge.open()">
  Подать лимит на аудит
</button>

<!-- 3. Передача предварительно выбранных данных (из калькулятора) -->
<script>
  function openAuditWithParams() {
    VIPConcierge.open({
      bk: 'Stake',                       // Букмекер
      tier: 'Platinum IV',               // VIP Ранг
      limit: '$25,000',                  // Лимит 1X2 на АПЛ/ЛЧ
      telegram: '@my_vip_handle'         // Опциональный контакт
    }, {
      tgDeskUsername: 'syndicate_vip_desk', // Telegram деск команды
      webhookUrl: '/api/audit-request'      // Эндпоинт отправки
    });
  }
</script>
